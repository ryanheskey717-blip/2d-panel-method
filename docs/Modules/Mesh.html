 49    def initialiseSetup(self, first=False):
 50        r"""Loads geometry, calculates panel properties, and preallocates arrays.
 51
 52        Scales the input vertices by the chord length, calculates normalized 
 53        tangent and normal vectors, determines control point locations, and 
 54        initializes the $A$ matrix and $b$ vector.
 55
 56        Args:
 57            first (bool, optional): Flag indicating if this is the first setup pass.
 58        """
 59        # ================ Start Everything ================ #
 60        # record start time and initialise
 61        self.start_time = time.time()
 62        self.last_time_check = 0
 63        self.converged = False
 64        self.iteration = 1
 65
 66        # load geometry
 67        self.vertices = np.loadtxt(self.config.geo_file, skiprows=1)
 68        self.vertices *= self.config.chord
 69
 70        # ============= Calculate control points, normals, tangents and lengths ============== #
 71        self.normals = []
 72        for i in range(len(self.vertices) - 1):
 73            self.normals.append([
 74                (self.vertices[i+1, 1] - self.vertices[i, 1]),
 75                -(self.vertices[i+1, 0] - self.vertices[i, 0]),
 76            ])
 77        self.normals = np.array(self.normals)
 78        self.normals /= np.linalg.norm(self.normals, axis=1, keepdims=True) # normalise
 79
 80        self.tangents = []
 81        for i in range(len(self.vertices) - 1):
 82            self.tangents.append([
 83                -(self.vertices[i+1, 0] - self.vertices[i, 0]),
 84                -(self.vertices[i+1, 1] - self.vertices[i, 1]),
 85            ])
 86        self.tangents = np.array(self.tangents)
 87        self.lengths = np.linalg.norm(self.tangents, axis=1)
 88        self.tangents /= np.linalg.norm(self.tangents, axis=1, keepdims=True) # normalise
 89
 90        self.control_points = []
 91        for i in range(len(self.vertices) - 1):
 92            self.control_points.append([
 93                (self.vertices[i, 0] + self.vertices[i+1, 0]) / 2, 
 94                (self.vertices[i, 1] + self.vertices[i+1, 1]) / 2
 95            ])
 96        self.control_points = np.array(self.control_points)
 97        self.control_points += self.normals * self.config.control_point_offset # offset a bit from surface
 98
 99        # ================= Preallocate Arrays ==================== #
100        self.N = len(self.control_points)
101        self.A = np.zeros((self.N+1, self.N+1)) # +1 for kutta condition
102        self.b = np.zeros(self.N+1)
103        self.delta_star = np.zeros(self.N) # BL computed at each control point
104        self.updateWakePoints(first=True)
105        self.transpiration_velocity = np.zeros(self.N)
106        self.normals_at_vertices = self.interpControlPointsToVertices(self.normals)
107        self.normals_at_vertices /= np.linalg.norm(self.normals_at_vertices, axis=1, keepdims=True)

109    # =================== Preconditioning =================== #
110    def computeAandb(self):
111        r"""Constructs the linear system for the panel method.
112
113        Populates the aerodynamic influence coefficient matrix $A$ and boundary 
114        condition vector $b$. Implements the zero-penetration (or transpiration) 
115        boundary condition across all control points and applies the Kutta 
116        condition at the trailing edge.
117        """
118        # section of A relating to no-penetration boundary condition
119        for i in range(self.N): # control point
120            for j in range(self.N + 1): # panel + influence from constant strength vortex sheet
121                if j == self.N: # vortex influence
122                    u, v = flows.getConstantVortexPanelsVelocity(self, self.control_points[i], strength=1.0)
123                    self.A[i, j] = np.dot(self.normals[i], np.array([u, v]))
124                else:
125                    if i == j: # source self-influence
126                        self.A[i, j] = 0.5
127                    else:
128                        u, v = flows.getConstantSourcePanelVelocity(self.vertices[j], self.vertices[j+1], self.control_points[i], strength=1.0)
129                        self.A[i, j] = np.dot(self.normals[i], np.array([u, v]))
130
131        # section of A relating to kutta condition (these find tangential influences)
132        for j in range(self.N + 1): # panel + influence from constant strength vortex sheet
133            if j == self.N: # vortex influence
134                u1, v1 = flows.getConstantVortexPanelsVelocity(self, self.control_points[0], strength=1.0)  # at upper TE control point
135                u2, v2 = flows.getConstantVortexPanelsVelocity(self, self.control_points[-1], strength=1.0) # at lower TE control point
136                self.A[-1, j] = (np.dot(self.tangents[0], np.array([u1, v1]))) + (np.dot(self.tangents[-1], np.array([u2, v2])))
137            else:
138                u1, v1 = flows.getConstantSourcePanelVelocity(self.vertices[j], self.vertices[j+1], self.control_points[0], strength=1.0)  # at upper TE control point
139                u2, v2 = flows.getConstantSourcePanelVelocity(self.vertices[j], self.vertices[j+1], self.control_points[-1], strength=1.0) # at lower TE control point
140                self.A[-1, j] = (np.dot(self.tangents[0], np.array([u1, v1]))) + (np.dot(self.tangents[-1], np.array([u2, v2])))
141
142        # RHS
143        for i in range(self.N + 1): # control points + kutta condition
144            if i == self.N: # kutta row
145                self.b[-1] = - (np.dot(self.config.V_inf_vec, self.tangents[0]) + np.dot(self.config.V_inf_vec, self.tangents[-1])) # - (upper + lower)
146            else:
147                self.b[i] = - np.dot(self.config.V_inf_vec, self.normals[i]) + self.transpiration_velocity[i]

151    def solveInviscid(self):
152        r"""Solves the linear system to find source strengths and vortex strength."""
153        x = np.linalg.solve(self.A, self.b)
154        self.source_strengths = x[:-1]
155        self.vortex_strength = x[-1]

180    def interpControlPointsToVertices(self, cp_vals):
181        r"""Interpolates values from panel control points to panel vertices.
182
183        Args:
184            cp_vals (ndarray): Array of scalar values evaluated at control points.
185
186        Returns:
187            ndarray: Array of values interpolated to the vertices using inverse distance weighting.
188        """
189        # find distances of vertices to control points
190        outs = []
191        dist1 = 0; dist2 = 0; v1 = 0; v2 = 0
192        for i in range(len(cp_vals)+1):
193            if i == 0: # first vertex
194                dist1 = self.lengths[-1] / 2
195                dist2 = self.lengths[i] / 2
196                v1 = cp_vals[-1]
197                v2 = cp_vals[i]
198            elif i == len(cp_vals): # last vertex
199                dist1 = self.lengths[i-1] / 2
200                dist2 = self.lengths[0] / 2
201                v1 = cp_vals[i-1]
202                v2 = cp_vals[0]
203            else:
204                dist1 = self.lengths[i-1] / 2
205                dist2 = self.lengths[i] / 2
206                v1 = cp_vals[i-1]
207                v2 = cp_vals[i]
208            w1 = 1 / dist1
209            w2 = 1 / dist2
210            outs.append((w1 * v1 + w2 * v2) / (w1 + w2))
211        outs = np.array(outs)
212        return outs

231    def calculateTangentialVelocityAtControlPoints(self):
232        r"""Evaluates the total tangential velocity at each control point."""
233        self.velocity_tangent_at_control_points = np.zeros(self.N)
234        for i in range(self.N): # control point
235            total_vel = self.config.V_inf_vec + flows.getConstantSourcePanelsVelocity(self, self.control_points[i]) + flows.getConstantVortexPanelsVelocity(self, self.control_points[i])
236            self.velocity_tangent_at_control_points[i] = np.dot(total_vel, self.tangents[i])

248    def pressureFromBernoulli(self): # dimensional
249        r"""Calculates dimensional pressure and the pressure coefficient $C_p$ using Bernoulli's equation."""
250        self.pressure = self.config.p_inf - 0.5 * self.config.rho_inf * ( self.velocity_tangent_at_control_points ** 2 - self.config.V_inf ** 2 )
251        self.cp = 1 - (self.velocity_tangent_at_control_points / self.config.V_inf) ** 2

253    def calculateForcesAndMoments(self):
254        r"""Calculates the total lift, drag, and moment coefficients from pressure and viscous forces."""
255        self.total_force = self.pressureForce() + self.viscousForce()
256        self.total_moment = self.pressureMoment() + self.viscousMoment()
257
258        self.c_force = self.total_force / ( 0.5 * self.config.rho_inf * self.config.V_inf ** 2 * self.config.chord)
259        self.c_moment = self.total_moment / ( 0.5 * self.config.rho_inf * self.config.V_inf ** 2 * self.config.chord ** 2)
260
261        self.lift = np.dot(self.total_force, np.array([-np.sin(self.config.alpha),np.cos(self.config.alpha)]))
262        self.drag = np.dot(self.total_force, np.array([np.cos(self.config.alpha), np.sin(self.config.alpha)]))
263        self.c_lift = np.dot(self.c_force, np.array([-np.sin(self.config.alpha), np.cos(self.config.alpha)]))
264        self.c_drag = np.dot(self.c_force, np.array([np.cos(self.config.alpha), np.sin(self.config.alpha)]))

266    def pressureForce(self): # sum integrate for forces (as only know pressure at control points i.e. centres)
267        r"""Integrates the pressure distribution to find the net inviscid force vector.
268
269        Returns:
270            ndarray: The $(F_x, F_y)$ force components generated by panel pressures.
271        """
272        fx = np.sum( - self.pressure * self.lengths * self.normals[:, 0] )
273        fy = np.sum( - self.pressure * self.lengths * self.normals[:, 1] )
274
275        return np.array([fx, fy])

277    def viscousForce(self): #### TODO: investigate issue of discontinuous force vs. aoa (i think due to i_stag changing)
278        r"""Integrates the skin friction distribution to find the net viscous force vector.
279
280        Returns:
281            ndarray: The $(F_x, F_y)$ force components generated by skin friction.
282        """
283        if self.config.include_viscous:
284            q = 0.5 * self.config.rho_inf * self.config.V_inf**2
285            fx = 0; fy = 0
286            for i in np.arange(self.i_stag, -1, -1): # top surface
287                force = q * self.lengths[i] * self.skin_friction_coeff[i]
288                fx += force * self.tangents[i, 0]
289                fy += force * self.tangents[i, 1]
290            for i in range(self.i_stag+1, self.N): # bottom surface
291                force = q * self.lengths[i] * self.skin_friction_coeff[i]
292                fx += force * -self.tangents[i, 0]
293                fy += force * -self.tangents[i, 1]
294
295            return np.array([fx, fy])
296        else:
297            return np.array([0, 0])

299    def pressureMoment(self):
300        r"""Calculates the pitching moment generated by pressure forces around the reference point.
301
302        Returns:
303            float: The net aerodynamic moment from inviscid pressure.
304        """
305        r = self.control_points - self.config.ref_position
306        forces = - self.pressure[:, None] * self.normals * self.lengths[:, None]
307
308        return np.sum(r[:, 0] * forces[:, 1] - r[:, 1] * forces[:, 0])
309
310    def viscousMoment(self):
311        r"""Calculates the pitching moment generated by viscous forces around the reference point.
312
313        Returns:
314            float: The net aerodynamic moment from skin friction.
315        """
316        if self.config.include_viscous:
317            r = self.control_points - self.config.ref_position
318            
319            q = 0.5 * self.config.rho_inf * self.config.V_inf**2
320            forces = np.zeros((self.N, 2))
321            for i in np.arange(self.i_stag, -1, -1): # top surface
322                force = q * self.lengths[i] * self.skin_friction_coeff[i]
323                forces[i, 0] += force * self.tangents[i, 0]
324                forces[i, 1] = force * self.tangents[i, 1]
325            for i in range(self.i_stag+1, self.N): # bottom surface
326                force = q * self.lengths[i] * self.skin_friction_coeff[i]
327                forces[i, 0] = force * -self.tangents[i, 0]
328                forces[i, 1] = force * -self.tangents[i, 1]
329
330            return np.sum(r[:, 0] * forces[:, 1] - r[:, 1] * forces[:, 0])
331        else:
332            return 0

 50    def initialiseSetup(self, first=False):
 51        r"""Loads geometry, calculates panel properties, and preallocates arrays.
 52
 53        Scales the input vertices by the chord length, calculates normalized 
 54        tangent and normal vectors, determines control point locations, and 
 55        initializes the $A$ matrix and $b$ vector.
 56
 57        Args:
 58            first (bool, optional): Flag indicating if this is the first setup pass.
 59        """
 60        # ================ Start Everything ================ #
 61        # record start time and initialise
 62        self.start_time = time.time()
 63        self.last_time_check = 0
 64        self.converged = False
 65        self.iteration = 1
 66
 67        # load geometry
 68        self.vertices = np.loadtxt(self.config.geo_file, skiprows=1)
 69        self.vertices *= self.config.chord
 70
 71        # ============= Calculate control points, normals, tangents and lengths ============== #
 72        self.normals = []
 73        for i in range(len(self.vertices) - 1):
 74            self.normals.append([
 75                (self.vertices[i+1, 1] - self.vertices[i, 1]),
 76                -(self.vertices[i+1, 0] - self.vertices[i, 0]),
 77            ])
 78        self.normals = np.array(self.normals)
 79        self.normals /= np.linalg.norm(self.normals, axis=1, keepdims=True) # normalise
 80
 81        self.tangents = []
 82        for i in range(len(self.vertices) - 1):
 83            self.tangents.append([
 84                -(self.vertices[i+1, 0] - self.vertices[i, 0]),
 85                -(self.vertices[i+1, 1] - self.vertices[i, 1]),
 86            ])
 87        self.tangents = np.array(self.tangents)
 88        self.lengths = np.linalg.norm(self.tangents, axis=1)
 89        self.tangents /= np.linalg.norm(self.tangents, axis=1, keepdims=True) # normalise
 90
 91        self.control_points = []
 92        for i in range(len(self.vertices) - 1):
 93            self.control_points.append([
 94                (self.vertices[i, 0] + self.vertices[i+1, 0]) / 2, 
 95                (self.vertices[i, 1] + self.vertices[i+1, 1]) / 2
 96            ])
 97        self.control_points = np.array(self.control_points)
 98        self.control_points += self.normals * self.config.control_point_offset # offset a bit from surface
 99
100        # ================= Preallocate Arrays ==================== #
101        self.N = len(self.control_points)
102        self.A = np.zeros((self.N+1, self.N+1)) # +1 for kutta condition
103        self.b = np.zeros(self.N+1)
104        self.delta_star = np.zeros(self.N) # BL computed at each control point
105        self.updateWakePoints(first=True)
106        self.transpiration_velocity = np.zeros(self.N)
107        self.normals_at_vertices = self.interpControlPointsToVertices(self.normals)
108        self.normals_at_vertices /= np.linalg.norm(self.normals_at_vertices, axis=1, keepdims=True)

110    # =================== Preconditioning =================== #
111    def computeAandb(self):
112        r"""Constructs the linear system for the panel method.
113
114        Populates the aerodynamic influence coefficient matrix $A$ and boundary 
115        condition vector $b$. Implements the zero-penetration (or transpiration) 
116        boundary condition across all control points and applies the Kutta 
117        condition at the trailing edge.
118        """
119        # section of A relating to no-penetration boundary condition
120        for i in range(self.N): # control point
121            for j in range(self.N + 1): # panel + influence from constant strength vortex sheet
122                if j == self.N: # vortex influence
123                    u, v = flows.getConstantVortexPanelsVelocity(self, self.control_points[i], strength=1.0)
124                    self.A[i, j] = np.dot(self.normals[i], np.array([u, v]))
125                else:
126                    if i == j: # source self-influence
127                        self.A[i, j] = 0.5
128                    else:
129                        u, v = flows.getConstantSourcePanelVelocity(self.vertices[j], self.vertices[j+1], self.control_points[i], strength=1.0)
130                        self.A[i, j] = np.dot(self.normals[i], np.array([u, v]))
131
132        # section of A relating to kutta condition (these find tangential influences)
133        for j in range(self.N + 1): # panel + influence from constant strength vortex sheet
134            if j == self.N: # vortex influence
135                u1, v1 = flows.getConstantVortexPanelsVelocity(self, self.control_points[0], strength=1.0)  # at upper TE control point
136                u2, v2 = flows.getConstantVortexPanelsVelocity(self, self.control_points[-1], strength=1.0) # at lower TE control point
137                self.A[-1, j] = (np.dot(self.tangents[0], np.array([u1, v1]))) + (np.dot(self.tangents[-1], np.array([u2, v2])))
138            else:
139                u1, v1 = flows.getConstantSourcePanelVelocity(self.vertices[j], self.vertices[j+1], self.control_points[0], strength=1.0)  # at upper TE control point
140                u2, v2 = flows.getConstantSourcePanelVelocity(self.vertices[j], self.vertices[j+1], self.control_points[-1], strength=1.0) # at lower TE control point
141                self.A[-1, j] = (np.dot(self.tangents[0], np.array([u1, v1]))) + (np.dot(self.tangents[-1], np.array([u2, v2])))
142
143        # RHS
144        for i in range(self.N + 1): # control points + kutta condition
145            if i == self.N: # kutta row
146                self.b[-1] = - (np.dot(self.config.V_inf_vec, self.tangents[0]) + np.dot(self.config.V_inf_vec, self.tangents[-1])) # - (upper + lower)
147            else:
148                self.b[i] = - np.dot(self.config.V_inf_vec, self.normals[i]) + self.transpiration_velocity[i]

181    def interpControlPointsToVertices(self, cp_vals):
182        r"""Interpolates values from panel control points to panel vertices.
183
184        Args:
185            cp_vals (ndarray): Array of scalar values evaluated at control points.
186
187        Returns:
188            ndarray: Array of values interpolated to the vertices using inverse distance weighting.
189        """
190        # find distances of vertices to control points
191        outs = []
192        dist1 = 0; dist2 = 0; v1 = 0; v2 = 0
193        for i in range(len(cp_vals)+1):
194            if i == 0: # first vertex
195                dist1 = self.lengths[-1] / 2
196                dist2 = self.lengths[i] / 2
197                v1 = cp_vals[-1]
198                v2 = cp_vals[i]
199            elif i == len(cp_vals): # last vertex
200                dist1 = self.lengths[i-1] / 2
201                dist2 = self.lengths[0] / 2
202                v1 = cp_vals[i-1]
203                v2 = cp_vals[0]
204            else:
205                dist1 = self.lengths[i-1] / 2
206                dist2 = self.lengths[i] / 2
207                v1 = cp_vals[i-1]
208                v2 = cp_vals[i]
209            w1 = 1 / dist1
210            w2 = 1 / dist2
211            outs.append((w1 * v1 + w2 * v2) / (w1 + w2))
212        outs = np.array(outs)
213        return outs

249    def pressureFromBernoulli(self): # dimensional
250        r"""Calculates dimensional pressure and the pressure coefficient $C_p$ using Bernoulli's equation."""
251        self.pressure = self.config.p_inf - 0.5 * self.config.rho_inf * ( self.velocity_tangent_at_control_points ** 2 - self.config.V_inf ** 2 )

254    def calculateForcesAndMoments(self):
255        r"""Calculates the total lift, drag, and moment coefficients from pressure and viscous forces."""
256        self.total_force = self.pressureForce() + self.viscousForce()
257        self.total_moment = self.pressureMoment() + self.viscousMoment()
258
259        self.c_force = self.total_force / ( 0.5 * self.config.rho_inf * self.config.V_inf ** 2 * self.config.chord)
260        self.c_moment = self.total_moment / ( 0.5 * self.config.rho_inf * self.config.V_inf ** 2 * self.config.chord ** 2)
261
262        self.lift = np.dot(self.total_force, np.array([-np.sin(self.config.alpha),np.cos(self.config.alpha)]))
263        self.drag = np.dot(self.total_force, np.array([np.cos(self.config.alpha), np.sin(self.config.alpha)]))
264        self.c_lift = np.dot(self.c_force, np.array([-np.sin(self.config.alpha), np.cos(self.config.alpha)]))
265        self.c_drag = np.dot(self.c_force, np.array([np.cos(self.config.alpha), np.sin(self.config.alpha)]))

267    def pressureForce(self): # sum integrate for forces (as only know pressure at control points i.e. centres)
268        r"""Integrates the pressure distribution to find the net inviscid force vector.
269
270        Returns:
271            ndarray: The $(F_x, F_y)$ force components generated by panel pressures.
272        """
273        fx = np.sum( - self.pressure * self.lengths * self.normals[:, 0] )
274        fy = np.sum( - self.pressure * self.lengths * self.normals[:, 1] )
275
276        return np.array([fx, fy])

278    def viscousForce(self): #### TODO: investigate issue of discontinuous force vs. aoa (i think due to i_stag changing)
279        r"""Integrates the skin friction distribution to find the net viscous force vector.
280
281        Returns:
282            ndarray: The $(F_x, F_y)$ force components generated by skin friction.
283        """
284        if self.config.include_viscous:
285            q = 0.5 * self.config.rho_inf * self.config.V_inf**2
286            fx = 0; fy = 0
287            for i in np.arange(self.i_stag, -1, -1): # top surface
288                force = q * self.lengths[i] * self.skin_friction_coeff[i]
289                fx += force * self.tangents[i, 0]
290                fy += force * self.tangents[i, 1]
291            for i in range(self.i_stag+1, self.N): # bottom surface
292                force = q * self.lengths[i] * self.skin_friction_coeff[i]
293                fx += force * -self.tangents[i, 0]
294                fy += force * -self.tangents[i, 1]
295
296            return np.array([fx, fy])
297        else:
298            return np.array([0, 0])

300    def pressureMoment(self):
301        r"""Calculates the pitching moment generated by pressure forces around the reference point.
302
303        Returns:
304            float: The net aerodynamic moment from inviscid pressure.
305        """
306        r = self.control_points - self.config.ref_position
307        forces = - self.pressure[:, None] * self.normals * self.lengths[:, None]
308
309        return np.sum(r[:, 0] * forces[:, 1] - r[:, 1] * forces[:, 0])

311    def viscousMoment(self):
312        r"""Calculates the pitching moment generated by viscous forces around the reference point.
313
314        Returns:
315            float: The net aerodynamic moment from skin friction.
316        """
317        if self.config.include_viscous:
318            r = self.control_points - self.config.ref_position
319            
320            q = 0.5 * self.config.rho_inf * self.config.V_inf**2
321            forces = np.zeros((self.N, 2))
322            for i in np.arange(self.i_stag, -1, -1): # top surface
323                force = q * self.lengths[i] * self.skin_friction_coeff[i]
324                forces[i, 0] += force * self.tangents[i, 0]
325                forces[i, 1] = force * self.tangents[i, 1]
326            for i in range(self.i_stag+1, self.N): # bottom surface
327                force = q * self.lengths[i] * self.skin_friction_coeff[i]
328                forces[i, 0] = force * -self.tangents[i, 0]
329                forces[i, 1] = force * -self.tangents[i, 1]
330
331            return np.sum(r[:, 0] * forces[:, 1] - r[:, 1] * forces[:, 0])
332        else:
333            return 0
